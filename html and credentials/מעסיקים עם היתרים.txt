    <script>
        let showOnlyUncalled = false;

        // טעינת נתונים שמורים (וידוא שסימונים והערות נשמרים בדפדפן)
        window.onload = function() {
            let rows = document.querySelectorAll("#leadsTable tbody tr");
            rows.forEach(row => {
                let id = row.getAttribute("data-id");
                
                // שיקום מצב התיבה (התקשרתי)
                let isChecked = localStorage.getItem("call_" + id) === "true";
                let checkbox = row.querySelector(".call-checkbox");
                checkbox.checked = isChecked;
                if (isChecked) row.classList.add("called-row");

                // שיקום הערות
                let savedNote = localStorage.getItem("note_" + id);
                if (savedNote) {
                    row.querySelector(".notes-input").value = savedNote;
                }
            });
        }

        function updateRowStatus(checkbox) {
            let row = checkbox.closest("tr");
            let id = row.getAttribute("data-id");
            if (checkbox.checked) {
                row.classList.add("called-row");
                localStorage.setItem("call_" + id, "true");
            } else {
                row.classList.remove("called-row");
                localStorage.setItem("call_" + id, "false");
            }
            applyFilters();
        }

        function saveNote(input) {
            let row = input.closest("tr");
            let id = row.getAttribute("data-id");
            localStorage.setItem("note_" + id, input.value);
        }

        function filterTable() {
            applyFilters();
        }

        function toggleUncalledFilter() {
            showOnlyUncalled = !showOnlyUncalled;
            let btn = document.getElementById("filterUncalledBtn");
            if (showOnlyUncalled) {
                btn.classList.add("active");
                btn.innerText = "מציג רק כאלה שעוד לא התקשרת (לחץ להצגת הכל)";
            } else {
                btn.classList.remove("active");
                btn.innerText = "הצג רק כאלה שעוד לא התקשרתי";
            }
            applyFilters();
        }

        function applyFilters() {
            let input = document.getElementById("searchInput");
            let filter = input.value.toLowerCase();
            let table = document.getElementById("leadsTable");
            let tr = table.getElementsByTagName("tr");

            for (let i = 1; i < tr.length; i++) {
                let row = tr[i];
                let td = row.getElementsByTagName("td");
                let isChecked = row.querySelector(".call-checkbox").checked;
                
                let rowContainsFilter = false;
                for (let j = 0; j < td.length; j++) {
                    if (td[j]) {
                        if (td[j].innerText.toLowerCase().indexOf(filter) > -1) {
                            rowContainsFilter = true;
                            break;
                        }
                    }
                }

                let matchesUncalled = !showOnlyUncalled || !isChecked;

                if (rowContainsFilter && matchesUncalled) {
                    row.style.display = "";
                } else {
                    row.style.display = "none";
                }
            }
        }
    </script>
